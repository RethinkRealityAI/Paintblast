# IWSDK Migration — Phase 1 (Foundation) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rewrite PaintBlast-MR on `@iwsdk/core` with feature parity to the current vanilla Three.js app, plus `XRAnchor`-anchored splatter pool, depth occlusion, and an `.mcp.json` wired so Claude Code can drive the IWSDK MCP servers.

**Architecture:** TypeScript + IWSDK ECS. Gameplay concepts (balls, targets, palette slots) are entities with data-only components; behavior lives in systems. Splatters live in a single pooled `InstancedMesh` owned by one `SplatterField` entity (not one entity per splat) to protect perf. Scene understanding converts detected planes/meshes into ECS entities that drive both physics colliders and target placement.

**Tech Stack:** `@iwsdk/core`, `@iwsdk/cli`, `@felixtz/iwsdk-rag-mcp`, `@meta-quest/hzdb`, Vite, Vitest, TypeScript, Havok (via IWSDK), UIKitML.

**Related spec:** `docs/superpowers/specs/2026-04-10-iwsdk-migration-design.md` (see Sections 1–4 + Section 7 for Phase 1 scope).

**Phases 2 and 3** (physics + game loop, voice/gesture/smoke test) will get their own plans after Phase 1's gate is green.

```typescript
describe('SplatterPool ring buffer', () => {
```

```typescript
// src/systems/SplatterSystem.ts
import * as THREE from 'three';
import { createSystem } from '@iwsdk/core';
import { SplatterField } from '../components/splatterField';

type Vec3 = [number, number, number];

/**
 * Ring-buffer splat pool. Pure data — no rendering — so it's unit-testable
 * without a Three.js context or an IWSDK World.
 * The SplatterSystem wraps one of these and binds it to an InstancedMesh.
 */
export class SplatterPool {
  readonly capacity: number;
  writeIdx = 0;
  liveCount = 0;

  private positions: Float32Array;
  private normals: Float32Array;
  private colors: Float32Array;

  constructor(capacity: number) {
    this.capacity = capacity;
    this.positions = new Float32Array(capacity * 3);
    this.normals = new Float32Array(capacity * 3);
    this.colors = new Float32Array(capacity * 3);
  }

  addSplat(position: Vec3, normal: Vec3, color: Vec3): number {
    const slot = this.writeIdx;
    const base = slot * 3;
    this.positions[base]     = position[0];
    this.positions[base + 1] = position[1];
    this.positions[base + 2] = position[2];
    this.normals[base]       = normal[0];
    this.normals[base + 1]   = normal[1];
    this.normals[base + 2]   = normal[2];
    this.colors[base]        = color[0];
    this.colors[base + 1]    = color[1];
    this.colors[base + 2]    = color[2];

    this.writeIdx = (this.writeIdx + 1) % this.capacity;
    if (this.liveCount < this.capacity) this.liveCount++;
    return slot;
  }

  clear(): void {
    this.writeIdx = 0;
    this.liveCount = 0;
    this.positions.fill(0);
    this.normals.fill(0);
    this.colors.fill(0);
  }

  getSplatPosition(slot: number): Vec3 {
    const b = slot * 3;
    return [this.positions[b], this.positions[b + 1], this.positions[b + 2]];
  }

  getSplatNormal(slot: number): Vec3 {
    const b = slot * 3;
    return [this.normals[b], this.normals[b + 1], this.normals[b + 2]];
  }

  getSplatColor(slot: number): Vec3 {
    const b = slot * 3;
    return [this.colors[b], this.colors[b + 1], this.colors[b + 2]];
  }
}

/**
 * System wrapper — owns the InstancedMesh and updates it from the pool.
 * Registered with the World; queries the single SplatterField entity.
 */
export class SplatterSystem extends createSystem({
  field: { required: [SplatterField] },
}) {
  private pool!: SplatterPool;
  private mesh!: THREE.InstancedMesh;

  init() {
    // Build the instanced mesh once.
    const geom = new THREE.CircleGeometry(0.08, 12);
    const mat = new THREE.MeshBasicMaterial({
      transparent: true,
      depthWrite: false,
      side: THREE.DoubleSide,
    });
    mat.userData.xrDepthOcclusion = true;

    const capacity = 256;
    this.mesh = new THREE.InstancedMesh(geom, mat, capacity);
    this.mesh.count = 0;
    this.mesh.frustumCulled = false;

    this.pool = new SplatterPool(capacity);

    // Attach the mesh to the SplatterField entity's object3D when it appears.
    this.queries.field.subscribe('qualify', (entity) => {
      entity.object3D.add(this.mesh);
    });
  }

  addSplat(worldPosition: Vec3, worldNormal: Vec3, color: Vec3): void {
    const slot = this.pool.addSplat(worldPosition, worldNormal, color);
    this.updateInstance(slot);
    this.mesh.count = this.pool.liveCount;
    this.mesh.instanceMatrix.needsUpdate = true;
  }

  clear(): void {
    this.pool.clear();
    this.mesh.count = 0;
    this.mesh.instanceMatrix.needsUpdate = true;
  }

  private updateInstance(slot: number): void {
    const pos = this.pool.getSplatPosition(slot);
    const nrm = this.pool.getSplatNormal(slot);
    const clr = this.pool.getSplatColor(slot);

    const m = new THREE.Matrix4();
    const q = new THREE.Quaternion().setFromUnitVectors(
      new THREE.Vector3(0, 0, 1),
      new THREE.Vector3(nrm[0], nrm[1], nrm[2]),
    );
    m.compose(new THREE.Vector3(pos[0], pos[1], pos[2]), q, new THREE.Vector3(1, 1, 1));
    this.mesh.setMatrixAt(slot, m);

    const c = new THREE.Color(clr[0], clr[1], clr[2]);
    this.mesh.setColorAt(slot, c);
    if (this.mesh.instanceColor) this.mesh.instanceColor.needsUpdate = true;
  }
}

type Vec3Input = Vec3;
```

```typescript
// src/systems/PaletteBindingSystem.ts
import { createSystem } from '@iwsdk/core';
import { PaletteSlot } from '../components/paletteSlot';
import type { UIStoreSlot } from '../types';
import { BallKind } from '../types';

interface SlotSnapshot {
  id: string;
  kind: BallKind;
  colorR: number;
  colorG: number;
  colorB: number;
}

export function groupSlotsIntoRows(
  slots: SlotSnapshot[],
  columns: number,
): UIStoreSlot[][] {
  const rows: UIStoreSlot[][] = [];
  let current: UIStoreSlot[] = [];
  for (const s of slots) {
    current.push({
      id: s.id,
      hex: rgbToHex(s.colorR, s.colorG, s.colorB),
      kind: s.kind,
    });
    if (current.length === columns) {
      rows.push(current);
      current = [];
    }
  }
  if (current.length > 0) rows.push(current);
  return rows;
}

function rgbToHex(r: number, g: number, b: number): string {
  const h = (v: number) =>
    Math.round(Math.max(0, Math.min(1, v)) * 255)
      .toString(16)
      .padStart(2, '0');
  return `#${h(r)}${h(g)}${h(b)}`;
}

export class PaletteBindingSystem extends createSystem({
  slots: { required: [PaletteSlot] },
}) {
  init() {
    const snapshots: SlotSnapshot[] = [];
    for (const entity of this.queries.slots.entities) {
      snapshots.push({
        id: String(entity.index),
        kind: entity.getValue(PaletteSlot, 'kind') as BallKind,
        colorR: entity.getValue(PaletteSlot, 'colorR')!,
        colorG: entity.getValue(PaletteSlot, 'colorG')!,
        colorB: entity.getValue(PaletteSlot, 'colorB')!,
      });
    }
    const rows = groupSlotsIntoRows(snapshots, 4);
    // Push into the UIKitML store. Exact path from api-notes.md.
    const store = (this.world as any).ui?.store;
    if (store) store.rows = rows;
  }
}
```
